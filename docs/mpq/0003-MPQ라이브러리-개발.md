# MPQ 라이브러리 개발 계획

선행 문서: `docs/mpq/0001-MPQ라이브러리-방안.md`, `docs/mpq/0002-MPQ라이브러리-설계.md`

## 공통

- 단위마다 구현 → 해당 단위 테스트 통과 → 다음 단위. 전체 일괄 테스트로 대체하지 않는다.
- 테스트 실행: `dotnet test tests/Biz.Bizadm.MPQ.NET.Tests --filter "FullyQualifiedName~<클래스>"`
- 코드 변경 시 `vibe-harness` 태스크 파일을 사용한다.
- 네임스페이스: `Biz.Bizadm.MPQ.NET`(공개), 내부는 `Biz.Bizadm.MPQ.NET.Internal`, 코덱은 `Biz.Bizadm.MPQ.NET.Compression`.

## 단위 3. StormCrypto

- **대상**: `src/.../Internal/StormCrypto.cs`
- **코드 초안**:

```csharp
internal static class StormCrypto
{
    private static readonly uint[] Table = BuildTable();

    private static uint[] BuildTable()
    {
        var table = new uint[0x500];
        uint seed = 0x00100001;
        for (int i = 0; i < 0x100; i++)
        for (int j = 0, index = i; j < 5; j++, index += 0x100)
        {
            seed = (seed * 125 + 3) % 0x2AAAAB;
            uint high = (seed & 0xFFFF) << 16;
            seed = (seed * 125 + 3) % 0x2AAAAB;
            table[index] = high | (seed & 0xFFFF);
        }
        return table;
    }

    public static uint HashString(ReadOnlySpan<char> name, MpqHashType type)
    {
        uint seed1 = 0x7FED7FED, seed2 = 0xEEEEEEEE;
        foreach (char c in name)
        {
            uint ch = NormalizeChar(c);  // ASCII 대문자, '/' → '\'
            seed1 = Table[((int)type << 8) + ch] ^ (seed1 + seed2);
            seed2 = ch + seed1 + seed2 + (seed2 << 5) + 3;
        }
        return seed1;
    }

    public static void DecryptBlock(Span<uint> data, uint key)
    {
        uint seed = 0xEEEEEEEE;
        for (int i = 0; i < data.Length; i++)
        {
            seed += Table[0x400 + (key & 0xFF)];
            uint plain = data[i] ^ (key + seed);
            key = ((~key << 0x15) + 0x11111111) | (key >> 0x0B);
            seed = plain + seed + (seed << 5) + 3;
            data[i] = plain;
        }
    }
    // EncryptBlock: 같은 흐름, seed 갱신에 암호화 전 평문 사용

    public static uint GetFileKey(string fileName, long blockOffset, uint fileSize, MpqFileFlags flags)
    {
        int slash = fileName.AsSpan().LastIndexOfAny('\\', '/');
        uint key = HashString(fileName.AsSpan(slash + 1), MpqHashType.FileKey);   // 경로를 뺀 파일명만
        return flags.HasFlag(MpqFileFlags.FixKey) ? (key + (uint)blockOffset) ^ fileSize : key;
    }
}
```

- 바이트 Span 버전은 `MemoryMarshal.Cast<byte, uint>`로 처리하고, 4바이트 미만 꼬리는 암호화하지 않는다(Storm 동작).
- **검증**: `StormCryptoTests` — `HashString("(hash table)", FileKey) == 0xC3AF3770`, `HashString("(block table)", FileKey) == 0xEC83B3A3`, 대소문자·`/` 정규화 동일 해시, Encrypt→Decrypt 라운드트립
- **선행 단위**: 1

```csharp
internal const uint HeaderSignature = 0x1A51504D;   // "MPQ\x1A"
internal const uint UserDataSignature = 0x1B51504D; // "MPQ\x1B"

public static (MpqHeader header, MpqUserData? userData, long headerOffset) Find(Stream s, MpqOpenOptions options)
{
    for (long pos = 0; pos + 32 <= s.Length; pos += 0x200)
    {
        uint sig = ReadUInt32At(s, pos);
        if (sig == UserDataSignature)
        {
            var ud = MpqUserData.Read(s, pos);  // UserDataSize, HeaderOffset, UserDataHeaderSize, Content
            long headerPos = pos + ud.HeaderOffset;
            if (ReadUInt32At(s, headerPos) == HeaderSignature)
                return (MpqHeader.Read(s, headerPos, options), ud, headerPos);
        }
        else if (sig == HeaderSignature)
            return (MpqHeader.Read(s, pos, options), null, pos);
    }
    throw new InvalidDataException("MPQ header not found.");
}
```

```csharp
internal readonly record struct MpqHashEntry(uint HashA, uint HashB, ushort Locale, ushort Platform, uint BlockIndex)
{
    public const uint Free = 0xFFFFFFFF, Deleted = 0xFFFFFFFE;
}

public static MpqHashEntry[] Read(Stream s, long position, uint count)
{
    var raw = new uint[count * 4];
    ReadExactly(s, position, MemoryMarshal.AsBytes(raw.AsSpan()));
    StormCrypto.DecryptBlock(raw, StormCrypto.HashString("(hash table)", MpqHashType.FileKey));
    // raw → MpqHashEntry[] (Locale/Platform은 3번째 uint의 하위/상위 16bit)
}

// MpqFileTable: 이름 조회
public MpqFileEntry? Find(string name, ushort locale = 0)
{
    uint start = StormCrypto.HashString(name, MpqHashType.TableOffset) & (uint)(_hash.Length - 1);
    uint a = StormCrypto.HashString(name, MpqHashType.NameA), b = StormCrypto.HashString(name, MpqHashType.NameB);
    for (uint i = start, n = 0; n < _hash.Length; i = (i + 1) & (uint)(_hash.Length - 1), n++)
    {
        ref readonly var h = ref _hash[i];
        if (h.BlockIndex == MpqHashEntry.Free) break;
        if (h.BlockIndex != MpqHashEntry.Deleted && h.HashA == a && h.HashB == b && h.Locale == locale)
            return _entries[(int)h.BlockIndex];
    }
    return null;
}
```

- `MpqFileFlags`: `Implode=0x100, Compress=0x200, Encrypted=0x10000, FixKey=0x20000, PatchFile=0x100000, SingleUnit=0x1000000, DeleteMarker=0x2000000, SectorCrc=0x4000000, Signature=0x10000000, Exists=0x80000000` (StormLib v9.40 `StormLib.h` 값과 대조함. `FixKey`는 StormLib의 `MPQ_FILE_KEY_V2`)
- **검증**: `MpqFileTableTests` — v1 샘플에서 expected 파일명 전부 `Find` 성공, 엔트리 크기·플래그 일치, 없는 이름은 null

```csharp
public void AddFile(string fileName, Stream content, MpqAddFileOptions? options = null)
{
    EnsureWritable();
    options ??= MpqAddFileOptions.Default;
    long position = _dataEnd;                         // 마지막 파일 데이터 끝(테이블 앞)
    var entry = MpqSectorWriter.Write(_stream, _headerOffset, position, fileName, content, options, _header.SectorSize);
    _table.Add(fileName, entry, options.Locale, overwrite: options.Overwrite);  // 해시 슬롯 확보, 가득 차면 예외
    _dataEnd = position + entry.CompressedSize;
    _dirty = true;
}

public void Flush()
{
    if (!_dirty) return;
    UpdateListFileAndAttributes();                    // 단위 11 직렬화 사용
    long tablePos = _dataEnd;
    MpqTableWriter.Write(_stream, _headerOffset, tablePos, _table, _header);  // Hash → Block 순, 암호화
    MpqHeaderWriter.Write(_stream, _headerOffset, _header with { HashTablePosition = ..., BlockTablePosition = ..., ArchiveSize = ... });
    _stream.SetLength(_headerOffset + archiveSize);
    _dirty = false;
}
```

```csharp
static bool VerifyPkcs1(byte[] signatureLe, RsaKey key, ReadOnlySpan<byte> digestInfo)
{
    var s = new BigInteger(signatureLe, isUnsigned: true, isBigEndian: false);
    byte[] em = BigInteger.ModPow(s, key.Exponent, key.Modulus).ToByteArray(isUnsigned: true, isBigEndian: true);
    // 00 01 FF..FF 00 || DigestInfo(MD5) || hash 패딩 검증
```

```csharp
// PTCH 헤더: 'PTCH' PatchDataSize SizeBeforePatch SizeAfterPatch | 'MD5_' BlockSize MD5Before MD5After | 'XFRM' XfrmSize Type('BSD0'|'COPY')
public Stream OpenFile(string fileName)
{
    byte[]? data = ReadBaseOrNull(fileName);
    foreach (var patch in _patchChain)                         // 연결 순서대로
        if (patch.TryFindPatch(_patchPrefix(patch) + fileName, out var p))
```
